## 发布文章

### 接口信息

**接口地址**: `POST /api/v1/cms/member/articles/{id}/publish/`  
**权限要求**: 需要Member用户认证，只能发布自己的文章  
**功能说明**: 将草稿或待审核状态的文章发布

### 请求头

```
Authorization: Bearer {token}
X-Tenant-ID: {tenant_id}
```

### 路径参数

| 参数 | 类型 | 必填 | 说明 |
|------|------|------|------|
| id | integer | 是 | 文章ID |

### 业务规则

- ✅ 只能发布状态为`draft`（草稿）或`pending`（待审核）的文章
- ✅ 发布后状态变为`published`（已发布）
- ✅ 自动设置`published_at`为当前时间
- ❌ 已发布或已归档的文章不能重复发布

### 请求示例

```javascript
// 基本发布
const publishArticle = async (articleId) => {
  const response = await fetch(
    `http://your-domain.com/api/v1/cms/member/articles/${articleId}/publish/`,
    {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${token}`,
        'X-Tenant-ID': '1'
      }
    }
  );
  
  const result = await response.json();
  
  if (result.success) {
    console.log('文章发布成功');
    return result.data;
  } else {
    throw new Error(result.message);
  }
};

// 带确认的发布
const publishArticleWithConfirm = async (articleId, articleTitle) => {
  if (!confirm(`确定要发布文章"${articleTitle}"吗？`)) {
    return false;
  }
  
  try {
    const response = await fetch(
      `http://your-domain.com/api/v1/cms/member/articles/${articleId}/publish/`,
      {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'X-Tenant-ID': '1'
        }
      }
    );
    
    const result = await response.json();
    
    if (result.success) {
      showSuccess('文章已发布');
      return result.data;
    } else {
      showError(result.message);
      return false;
    }
  } catch (error) {
    showError('发布失败，请稍后重试');
    console.error('发布文章错误:', error);
    return false;
  }
};

// 发布前检查
const publishArticleWithValidation = async (article) => {
  // 检查文章状态
  if (article.status === 'published') {
    showError('文章已经发布过了');
    return false;
  }
  
  if (!['draft', 'pending'].includes(article.status)) {
    showError('只能发布草稿或待审核状态的文章');
    return false;
  }
  
  // 检查必填字段
  if (!article.title || !article.content) {
    showError('文章标题和内容不能为空');
    return false;
  }
  
  // 执行发布
  try {
    const response = await fetch(
      `http://your-domain.com/api/v1/cms/member/articles/${article.id}/publish/`,
      {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'X-Tenant-ID': '1'
        }
      }
    );
    
    const result = await response.json();
    
    if (result.success) {
      showSuccess('文章发布成功');
      // 更新本地文章状态
      article.status = 'published';
      article.published_at = result.data.published_at;
      return true;
    } else {
      showError(result.message);
      return false;
    }
  } catch (error) {
    showError('发布失败，请稍后重试');
    console.error('发布文章错误:', error);
    return false;
  }
};
```

### UI交互建议

```javascript
// 发布按钮状态管理
const ArticlePublishButton = ({ article, onPublished }) => {
  const [isPublishing, setIsPublishing] = useState(false);
  
  // 检查是否可以发布
  const canPublish = ['draft', 'pending'].includes(article.status);
  
  const handlePublish = async () => {
    if (!canPublish) {
      alert('当前状态不能发布');
      return;
    }
    
    if (!confirm('确定要发布这篇文章吗？')) {
      return;

    try {
      const result = await publishArticle(article.id);
      onPublished(result);
      showSuccess('文章已发布');
    } catch (error) {
      showError(error.message);
    } finally {
      setIsPublishing(false);
    }
  };
  
  return (
    <button 
      onClick={handlePublish}
      disabled={!canPublish || isPublishing}
      className={canPublish ? 'btn-primary' : 'btn-disabled'}
    >
      {isPublishing ? '发布中...' : '发布文章'}
    </button>
  );
};
```

## 获取文章统计

### 接口信息

**接口地址**: `GET /api/v1/cms/member/articles/{id}/statistics/`  
**权限要求**: 需要Member用户认证，只能查看自己的文章统计  
**功能说明**: 获取文章的浏览量、点赞数、评论数等统计信息

### 请求头

```
Authorization: Bearer {token}
X-Tenant-ID: {tenant_id}
```

### 路径参数

| 参数 | 类型 | 必填 | 说明 |
|------|------|------|------|
| id | integer | 是 | 文章ID |

### 请求示例

```javascript
// 获取单篇文章统计
const getArticleStatistics = async (articleId) => {
  const response = await fetch(
    `http://your-domain.com/api/v1/cms/member/articles/${articleId}/statistics/`,
    {
      headers: {
        'Authorization': `Bearer ${token}`,
        'X-Tenant-ID': '1'
      }
    }
  );
  
  const result = await response.json();
  
  if (result.success) {
    return result.data;
  } else {
    throw new Error(result.message);
  }
};

// 批量获取多篇文章统计
const getMultipleArticleStatistics = async (articleIds) => {
  const promises = articleIds.map(id => getArticleStatistics(id));
  
  try {
    const results = await Promise.all(promises);
    return results;
  } catch (error) {
    console.error('获取统计失败:', error);
    return [];
  }
};

// 获取并展示统计信息
const displayArticleStatistics = async (articleId) => {
  try {
    const stats = await getArticleStatistics(articleId);
    
    console.log('文章统计:');

    console.log(`- 独立访客: ${stats.unique_views_count}`);

    return stats;
  } catch (error) {

  }
};
```

## 文章状态流转

### 状态说明

文章有以下4种状态：

| 状态 | 值 | 说明 | 可见性 |
|------|---|------|--------|
| 草稿 | draft | 编辑中的文章 | 仅作者可见 |
| 待审核 | pending | 提交审核的文章 | 仅作者和管理员可见 |
| 已发布 | published | 正式发布的文章 | 所有人可见（根据visibility设置） |
| 已归档 | archived | 已删除/下线的文章 | 仅作者和管理员可见 |

### 状态流转图

```
         创建文章
            ↓
         [draft]
            ↓
        ├─ 发布 ─→ [published]
        │             ↓
        └─ 删除 ─→ [archived]
```

### 状态转换规则

| 当前状态 | 可转换到 | 操作方式 | 接口 |
|----------|----------|----------|------|
| draft | published | 发布 | POST /articles/{id}/publish/ |
| draft | archived | 删除 | DELETE /articles/{id}/ |
| pending | published | 发布 | POST /articles/{id}/publish/ |
| pending | archived | 删除 | DELETE /articles/{id}/ |
| published | archived | 删除 | DELETE /articles/{id}/ |
| archived | - | 不可转换 | - |

### 状态转换代码示例

```javascript
// 文章状态管理器
class ArticleStatusManager {
  constructor(article) {
    this.article = article;
  }
  
  // 检查是否可以发布
  canPublish() {
    return ['draft', 'pending'].includes(this.article.status);
  }
  
  // 检查是否可以删除
  canDelete() {
    return ['draft', 'pending', 'published'].includes(this.article.status);
  }
  
  // 检查是否可以编辑
  canEdit() {
    return this.article.status !== 'archived';
  }
  
  // 获取状态显示文本
  getStatusText() {
    const statusMap = {
      draft: '草稿',
      pending: '待审核',
      published: '已发布',
      archived: '已归档'
    };
    return statusMap[this.article.status] || '未知';
  }
  
  // 获取状态样式类
  getStatusClass() {
    const classMap = {
      draft: 'status-draft',
      pending: 'status-pending',
      published: 'status-published',
      archived: 'status-archived'
    };
    return classMap[this.article.status] || 'status-unknown';
  }
  
  // 获取可用操作
  getAvailableActions() {
    const actions = [];
    
    if (this.canEdit()) {
      actions.push({ key: 'edit', label: '编辑' });
    }
    
    if (this.canPublish()) {
      actions.push({ key: 'publish', label: '发布' });
    }
    
    if (this.canDelete()) {
      actions.push({ key: 'delete', label: '删除' });
    }
    
    return actions;
  }
}

// 使用示例
const article = { id: 12345, status: 'draft', title: '...' };
const manager = new ArticleStatusManager(article);

console.log('状态:', manager.getStatusText());
console.log('可以发布:', manager.canPublish());
console.log('可用操作:', manager.getAvailableActions());
```
