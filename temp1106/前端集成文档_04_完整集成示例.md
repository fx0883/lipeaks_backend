## 完整API客户端实现

### 基础API客户端

```javascript
// api-client.js
class MemberArticleAPI {
  constructor(config) {
    this.baseURL = config.baseURL || 'http://localhost:8000';
    this.token = null;
    this.tenantId = config.tenantId;
  }
  
  // 设置Token
  setToken(token) {
    this.token = token;
  }
  
  // 设置租户ID
  setTenantId(tenantId) {
    this.tenantId = tenantId;
  }
  
  // 通用请求方法
  async request(endpoint, options = {}) {
    const url = `${this.baseURL}${endpoint}`;
    const headers = {
      'Content-Type': 'application/json',
      ...options.headers
    };
    
    if (this.token) {
      headers['Authorization'] = `Bearer ${this.token}`;
    }
    
    if (this.tenantId) {
      headers['X-Tenant-ID'] = String(this.tenantId);
    }
    
    const response = await fetch(url, {
      ...options,
      headers
    });
    
    if (response.status === 204) {
      return null;
    }
    
    const result = await response.json();
    
    if (!result.success) {
      throw new APIError(result.message, result.code, result.error_code);
    }
    
    return result.data;
  }
  
  // 认证相关
  async login(username, password) {
    const data = await this.request('/api/v1/auth/login/', {
      method: 'POST',
      body: JSON.stringify({ username, password })
    });
    
    this.setToken(data.token);
    return data;
  }
  
  // 文章CRUD
  async getArticles(params = {}) {
    // 🚀 v2.0 新增：支持高性能查询参数
    // 推荐使用 member_id/user_id 获得最佳性能
    const query = new URLSearchParams(params).toString();
    return this.request(`/api/v1/cms/member/articles/?${query}`);
  }

  // 🚀 v2.0 新增：高性能查询方法
  async getMyArticles(options = {}) {
    // 推荐：使用 member_id 精确查询，性能最佳
    const params = {
      member_id: this.currentUserId, // 从登录状态获取
      ...options
    };
    return this.getArticles(params);
  }

  async getArticlesByAuthor(authorId, authorType, options = {}) {
    // 🚀 v2.0 优化：使用精确参数而非通用参数
    const params = { ...options };

    if (authorType === 'member') {
      params.member_id = authorId; // 高性能查询
    } else if (authorType === 'admin') {
      params.user_id = authorId; // 高性能查询
    } else {
      // 向后兼容：使用通用参数
      params.author_id = authorId;
    }

    return this.getArticles(params);
  }
  
  async getArticle(id) {
    return this.request(`/api/v1/cms/member/articles/${id}/`);
  }
  
  async createArticle(data) {
    return this.request('/api/v1/cms/member/articles/', {
      method: 'POST',
      body: JSON.stringify(data)
    });
  }
  
  async updateArticle(id, data) {
    return this.request(`/api/v1/cms/member/articles/${id}/`, {
      method: 'PATCH',
      body: JSON.stringify(data)
    });
  }
  
  async deleteArticle(id) {
    return this.request(`/api/v1/cms/member/articles/${id}/`, {
      method: 'DELETE'
    });
  }
  
  async publishArticle(id) {
    return this.request(`/api/v1/cms/member/articles/${id}/publish/`, {
      method: 'POST'
    });
  }
  
  async getArticleStatistics(id) {
    return this.request(`/api/v1/cms/member/articles/${id}/statistics/`);
  }
}

// 自定义错误类
class APIError extends Error {
  constructor(message, code, errorCode) {
    super(message);
    this.name = 'APIError';
    this.code = code;
    this.errorCode = errorCode;
  }
}

// 导出实例

  baseURL: process.env.REACT_APP_API_URL,
  tenantId: 1
```

### 1. 文章列表页面

```jsx
// ArticleList.jsx
import React, { useState, useEffect } from 'react';
import { articleAPI } from './api-client';

const ArticleList = () => {
  const [articles, setArticles] = useState([]);
  const [loading, setLoading] = useState(true);
  const [pagination, setPagination] = useState({
    page: 1,
    pageSize: 20,
    total: 0
  });
  
  useEffect(() => {
    fetchArticles();
  }, [pagination.page]);
  
  const fetchArticles = async () => {
    setLoading(true);
    try {
      const data = await articleAPI.getArticles({
        page: pagination.page,
        page_size: pagination.pageSize
      });
      
      setArticles(data.results);
      setPagination(prev => ({
        ...prev,
        total: data.count
      }));
    } catch (error) {
      console.error('获取文章列表失败:', error);
      alert(error.message);
    } finally {
      setLoading(false);
    }
  };
  
  const handleDelete = async (id) => {
    if (!confirm('确定要删除这篇文章吗？')) return;
    
    try {
      await articleAPI.deleteArticle(id);
      alert('删除成功');
      fetchArticles();
    } catch (error) {
      alert('删除失败: ' + error.message);
    }
  };
  
  const handlePublish = async (id) => {
    try {
      await articleAPI.publishArticle(id);
      alert('发布成功');
      fetchArticles();
    } catch (error) {
      alert('发布失败: ' + error.message);
    }
  };
  
  if (loading) {
    return <div>加载中...</div>;
  }
  
  return (
    <div className="article-list">
      <h1>我的文章</h1>
      
      <button onClick={() => window.location.href = '/articles/new'}>
        创建新文章
      </button>
      
      {articles.map(article => (
        <div key={article.id} className="article-item">
          <h3>{article.title}</h3>
          <p>{article.excerpt}</p>
          <div className="article-meta">
            <span>状态: {article.status}</span>
            <span>创建时间: {new Date(article.created_at).toLocaleString()}</span>
          </div>
          <div className="article-actions">
            <button onClick={() => window.location.href = `/articles/${article.id}/edit`}>
              编辑
            </button>
            {['draft', 'pending'].includes(article.status) && (
              <button onClick={() => handlePublish(article.id)}>
                发布
              </button>
            )}
            <button onClick={() => handleDelete(article.id)}>
              删除
            </button>
          </div>
        </div>
      ))}
      
      {/* 分页 */}
      <div className="pagination">
        <button 
          disabled={pagination.page === 1}
          onClick={() => setPagination(prev => ({ ...prev, page: prev.page - 1 }))}
        >
          上一页
        </button>
        <span>第 {pagination.page} 页</span>
        <button 
          disabled={pagination.page * pagination.pageSize >= pagination.total}
          onClick={() => setPagination(prev => ({ ...prev, page: prev.page + 1 }))}
        >
          下一页
        </button>
      </div>
    </div>
  );
};

export default ArticleList;
```

### 2. 文章编辑器

```jsx
// ArticleEditor.jsx
import React, { useState, useEffect } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { articleAPI } from './api-client';

const ArticleEditor = () => {
  const { id } = useParams();
  const navigate = useNavigate();
  const [article, setArticle] = useState({
    title: '',
    content: '',
    excerpt: '',
    status: 'draft',
    visibility: 'public',
    allow_comment: true
  });
  const [saving, setSaving] = useState(false);
  
  useEffect(() => {
    if (id) {
      fetchArticle();
    }
  }, [id]);
  
  const fetchArticle = async () => {
    try {
      const data = await articleAPI.getArticle(id);
      setArticle(data);
    } catch (error) {
      alert('获取文章失败: ' + error.message);
      navigate('/articles');
    }
  };
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    setSaving(true);
    
    try {
      if (id) {
        await articleAPI.updateArticle(id, article);
        alert('保存成功');
      } else {
        const newArticle = await articleAPI.createArticle(article);
        alert('创建成功');
        navigate(`/articles/${newArticle.id}/edit`);
      }
    } catch (error) {
      alert('保存失败: ' + error.message);
    } finally {
      setSaving(false);
    }
  };
  
  return (
    <div className="article-editor">
      <h1>{id ? '编辑文章' : '创建新文章'}</h1>
      
      <form onSubmit={handleSubmit}>
        <div className="form-group">
          <label>标题</label>
          <input
            type="text"
            value={article.title}
            onChange={e => setArticle({ ...article, title: e.target.value })}
            required
          />
        </div>
        
        <div className="form-group">
          <label>内容</label>
          <textarea
            value={article.content}
            onChange={e => setArticle({ ...article, content: e.target.value })}
            rows={20}
            required
          />
        </div>
        
        <div className="form-group">
          <label>摘要</label>
          <textarea
            value={article.excerpt}
            onChange={e => setArticle({ ...article, excerpt: e.target.value })}
            rows={3}
          />
        </div>
        
        <div className="form-group">
          <label>
            <input
              type="checkbox"
              checked={article.allow_comment}
              onChange={e => setArticle({ ...article, allow_comment: e.target.checked })}
            />
            允许评论
          </label>
        </div>
        
        <div className="form-actions">
          <button type="submit" disabled={saving}>
            {saving ? '保存中...' : '保存'}
          </button>
          <button type="button" onClick={() => navigate('/articles')}>
            取消
          </button>
        </div>
      </form>
    </div>
  );
};

export default ArticleEditor;
```

```vue
import { articleAPI } from './api-client';

      articles: [],
      loading: false,
      currentPage: 1,
      pageSize: 20,
      totalCount: 0,
      searchKeyword: ''
    };
  },

    totalPages() {
      return Math.ceil(this.totalCount / this.pageSize);
```

## 错误处理最佳实践

### 统一错误处理

```javascript
// error-handler.js
export class ErrorHandler {
  static handle(error) {
    if (error instanceof APIError) {
      return this.handleAPIError(error);
    }
    
    if (error instanceof NetworkError) {
      return this.handleNetworkError(error);
    }
    
    return this.handleUnknownError(error);
  }
  
  static handleAPIError(error) {
    switch (error.errorCode) {
      case 'AUTH_TOKEN_EXPIRED':
        // 跳转到登录页
        window.location.href = '/login?expired=1';
        break;
        
      case 'AUTH_PERMISSION_DENIED':
        alert('您没有执行此操作的权限');
        break;
        
      case 'VALIDATION_ERROR':
        alert('输入数据有误: ' + error.message);
        break;
        
      default:
        alert('操作失败: ' + error.message);
    }
  }
  
  static handleNetworkError(error) {
    alert('网络连接失败，请检查您的网络');
  }
  
  static handleUnknownError(error) {
    console.error('未知错误:', error);
    alert('发生了未知错误，请稍后重试');
  }
}
```

### 3. 数据缓存

```javascript
class CachedArticleAPI extends MemberArticleAPI {
  constructor(config) {
    super(config);
    this.cache = new Map();
    this.cacheTimeout = 5 * 60 * 1000; // 5分钟
  }
  
  async getArticle(id) {
    const cacheKey = `article_${id}`;
    const cached = this.cache.get(cacheKey);
    
    if (cached && Date.now() - cached.timestamp < this.cacheTimeout) {
      return cached.data;
    }
    
    const data = await super.getArticle(id);
    this.cache.set(cacheKey, {
      data,
      timestamp: Date.now()
    });
    
    return data;
  }
  
  clearCache(id) {
    if (id) {
      this.cache.delete(`article_${id}`);
    } else {
      this.cache.clear();
    }
  }
}
```

## 常见问题FAQ

### 🚀 v2.0 架构升级相关问题

#### Q0.1: 这次架构升级对我现有的代码有影响吗？
**A**: 完全没有影响！API响应格式100%保持兼容，所有现有代码无需修改即可获得性能提升。

#### Q0.2: 如何获得最佳性能？
**A**: 使用新的精确查询参数：
```javascript
// 高性能查询
await api.getArticles({ member_id: userId, status: 'published' });
await api.getArticles({ user_id: adminId, status: 'published' });

// 替代低性能查询
// await api.getArticles({ author_id: userId, status: 'published' }); // 较慢
```

#### Q0.3: 为什么查询速度这么快？
**A**: 后端从GenericForeignKey升级到双外键架构，消除了N+1查询问题，将数据库查询次数从11+次降低到1次。

### 常见集成问题

### Q2: 如何处理租户ID？

### Q3: 文章列表如何实现无限滚动？
